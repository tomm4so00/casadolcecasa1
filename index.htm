    function setupControls() {

      window.addEventListener('keydown', (e) => {

        let k = e.key.toLowerCase();

        if (k === 'w' || e.code === 'ArrowUp') keysPressed.w = true;

        if (k === 's' || e.code === 'ArrowDown') keysPressed.s = true;

        if (k === 'a' || e.code === 'ArrowLeft') keysPressed.a = true;

        if (k === 'd' || e.code === 'ArrowRight') keysPressed.d = true;

        updateKeyboardMovement();

      });

 

      window.addEventListener('keyup', (e) => {

        let k = e.key.toLowerCase();

        if (k === 'w' || e.code === 'ArrowUp') keysPressed.w = false;

        if (k === 's' || e.code === 'ArrowDown') keysPressed.s = false;

        if (k === 'a' || e.code === 'ArrowLeft') keysPressed.a = false;

        if (k === 'd' || e.code === 'ArrowRight') keysPressed.d = false;

        updateKeyboardMovement();

      });

 

      const leftZone = document.getElementById('left-touch-zone');

      const joystickEl = document.getElementById('dynamic-joystick');

      const knobEl = document.getElementById('joystick-knob');

 

      leftZone.addEventListener('touchstart', (e) => {

        for (let touch of e.changedTouches) {

          if (joystickTouchId === null) {

            joystickTouchId = touch.identifier;

            joystickOrigin = { x: touch.clientX, y: touch.clientY };

            joystickEl.style.left = `${touch.clientX}px`;

            joystickEl.style.top = `${touch.clientY}px`;

            joystickEl.style.display = 'block';

          }

        }

      });

 

      leftZone.addEventListener('touchmove', (e) => {

        for (let touch of e.changedTouches) {

          if (touch.identifier === joystickTouchId) {

            let dx = touch.clientX - joystickOrigin.x;

            let dy = touch.clientY - joystickOrigin.y;

            let dist = Math.hypot(dx, dy);

            let maxR = 40;

            if (dist > maxR) { dx = (dx/dist)*maxR; dy = (dy/dist)*maxR; }

            knobEl.style.transform = `translate(calc(-50% + ${dx}px), calc(-50% + ${dy}px))`;

            moveRight = dx / maxR;

            moveForward = -dy / maxR;

          }

        }

      });

 

      const resetJoy = (e) => {

        for (let touch of e.changedTouches) {

          if (touch.identifier === joystickTouchId) {

            joystickTouchId = null;

            if (!isKeyActive()) { moveForward = 0; moveRight = 0; }

            joystickEl.style.display = 'none';

          }

        }

      };

      leftZone.addEventListener('touchend', resetJoy);

      leftZone.addEventListener('touchcancel', resetJoy);

    }

 

    function isKeyActive() {

    function updateKeyboardMovement() {

      let fwd = 0, rt = 0;

      if (keysPressed.w) fwd += 1;

      if (keysPressed.s) fwd -= 1;

      if (keysPressed.a) rt -= 1;

      if (keysPressed.d) rt += 1;

 

      if (fwd !== 0 || rt !== 0) {

        moveForward = fwd;

        moveRight = rt;

      } else if (joystickTouchId === null) {

        moveForward = 0;

        moveRight = 0;

      }

    }

    function isoProject(x, y, z) {

      const cx = canvas.width / 2;

      const cy = canvas.height / 2 + 20;

      let isoX = (x - y) * 0.866;

      let isoY = (x + y) * 0.5 - z;

      return { x: cx + isoX, y: cy + isoY };

    }

 

    function drawIsoBlock(x, y, w, h, zHeight, mainColor, topColor, label) {

      let p1 = isoProject(x - w/2, y - h/2, 0);

      let p2 = isoProject(x + w/2, y - h/2, 0);

      let p3 = isoProject(x + w/2, y + h/2, 0);

      let p4 = isoProject(x - w/2, y + h/2, 0);

 

      let t1 = isoProject(x - w/2, y - h/2, zHeight);

      let t2 = isoProject(x + w/2, y - h/2, zHeight);

      let t3 = isoProject(x + w/2, y + h/2, zHeight);

      let t4 = isoProject(x - w/2, y + h/2, zHeight);

 

      ctx.strokeStyle = "rgba(0,0,0,0.3)";

      ctx.lineWidth = 1;

 

      ctx.fillStyle = mainColor;

      ctx.beginPath();

      ctx.moveTo(p1.x, p1.y); ctx.lineTo(p4.x, p4.y);

      ctx.lineTo(t4.x, t4.y); ctx.lineTo(t1.x, t1.y);

      ctx.fill(); ctx.stroke();

 

      ctx.fillStyle = mainColor;

      ctx.beginPath();

      ctx.moveTo(p4.x, p4.y); ctx.lineTo(p3.x, p3.y);

      ctx.lineTo(t3.x, t3.y); ctx.lineTo(t4.x, t4.y);

      ctx.fill(); ctx.stroke();

 

      ctx.fillStyle = topColor;

      ctx.beginPath();

      ctx.moveTo(t1.x, t1.y); ctx.lineTo(t2.x, t2.y);

      ctx.lineTo(t3.x, t3.y); ctx.lineTo(t4.x, t4.y);

      ctx.fill(); ctx.stroke();

 

      if (label) {

        ctx.fillStyle = "#fff";

        ctx.font = "11px sans-serif";

        ctx.textAlign = "center";

        ctx.fillText(label, t4.x, t4.y - 8);

      }

    }

 

    function drawCylinder(x, y, radius, zHeight, sides, sideColor, topColor) {

      let bottomPts = [], topPts = [];

      for (let i = 0; i < sides; i++) {

        let angle = (i / sides) * Math.PI * 2;

        let px = x + Math.cos(angle) * radius;

        let py = y + Math.sin(angle) * radius;

        bottomPts.push(isoProject(px, py, 0));

        topPts.push(isoProject(px, py, zHeight));

      }

 

      ctx.strokeStyle = "rgba(0,0,0,0.2)";

      ctx.lineWidth = 1;

 

      for (let i = 0; i < sides; i++) {

        let next = (i + 1) % sides;

        ctx.fillStyle = sideColor;

        ctx.beginPath();

        ctx.moveTo(bottomPts[i].x, bottomPts[i].y);

        ctx.lineTo(bottomPts[next].x, bottomPts[next].y);

        ctx.lineTo(topPts[next].x, topPts[next].y);

        ctx.lineTo(topPts[i].x, topPts[i].y);

        ctx.closePath();

        ctx.fill(); ctx.stroke();

      }

 

      ctx.fillStyle = topColor;

      ctx.beginPath();

      ctx.moveTo(topPts[0].x, topPts[0].y);

      for (let i = 1; i < sides; i++) {

        ctx.lineTo(topPts[i].x, topPts[i].y);

      }

      ctx.closePath();

      ctx.fill(); ctx.stroke();

    }

    function renderEnvironment(delta, animTime) {

      const speed = isSprinting ? 220 : 130;

      let map = getCurrentFurnitureMap();

 

      if (!activeEnvironment.startsWith('ANIM_')) {

        if (isMovingFurnitureMode && movingPropKey && map[movingPropKey]) {

          let item = map[movingPropKey];

          item.x += moveRight * speed * delta;

          item.y -= moveForward * speed * delta;

        } else if (!isSleeping) {

          playerPos.x += moveRight * speed * delta;

          playerPos.y -= moveForward * speed * delta;

          playerPos.x = Math.max(-170, Math.min(170, playerPos.x));

          playerPos.y = Math.max(-150, Math.min(150, playerPos.y));

        }

      } else {

        playerPos.x += moveRight * speed * delta;

        playerPos.y -= moveForward * speed * delta;

        playerPos.x = Math.max(-170, Math.min(170, playerPos.x));

        playerPos.y = Math.max(-150, Math.min(150, playerPos.y));

      }

 

      ctx.strokeStyle = "rgba(0,0,0,0.3)";

      ctx.lineWidth = 1;

 

      let p1 = isoProject(-200, -180, 0);

      let p2 = isoProject(200, -180, 0);

      let p3 = isoProject(200, 180, 0);

      let p4 = isoProject(-200, 180, 0);

 

      ctx.fillStyle = (activeEnvironment === 'SALOON') ? "#34495e" : "#8e5d38";

      ctx.beginPath();

      ctx.moveTo(p1.x, p1.y); ctx.lineTo(p2.x, p2.y);

      ctx.lineTo(p3.x, p3.y); ctx.lineTo(p4.x, p4.y);

      ctx.fill(); ctx.stroke();

 

      ctx.fillStyle = "#fff";

      ctx.font = "bold 15px sans-serif";

      ctx.textAlign = "center";

      ctx.fillText(currentRoomTitle, canvas.width / 2, 35);

 

      if (!activeEnvironment.startsWith('ANIM_')) {

        let targetAnim = fridgeIsOpen ? 1 : 0;

        fridgeAnimProgress += (targetAnim - fridgeAnimProgress) * (delta * 6);

 

        Object.keys(map).forEach(k => {

          let item = map[k];

          drawIsoBlock(item.x, item.y, item.w, item.h, item.z || 110, item.color || "#e74c3c", item.topColor || "#ff6b6b", item.name);

        });

      }

 

      // Renderizza tutti i giocatori connessi in tempo reale che si trovano nel tuo stesso ambiente

      Object.keys(onlinePlayers).forEach(id => {

        let p = onlinePlayers[id];

        if (p.env === activeEnvironment) {

          if (id === myPlayerId) {

            drawPlayerAvatar(playerPos.x, playerPos.y, p.name, animTime, true);

          } else {

            drawPlayerAvatar(p.x, p.y, p.name, animTime, false);

          }

        }

      });

 

      checkProximity();

    }

 

    let lastTime = performance.now();

    let animClock = 0;

 

    function startLoop() {

      function animate(now) {

        requestAnimationFrame(animate);

        let delta = (now - lastTime) / 1000;

        lastTime = now;

        if (isNaN(delta) || delta > 0.1) delta = 0.016;

        animClock += delta;

 

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        renderEnvironment(delta, animClock);

      }

      requestAnimationFrame(animate);

    }
